  def get_json(url: str, label: str) -> dict:
      print(f"GET {label}: {url}")
      try:
          with urllib.request.urlopen(url, timeout=90) as resp:
              return json.load(resp)
      except Exception as exc:
          print(f"Failed endpoint {label}: {url}")
          raise AssertionError(f"deployed smoke failed at {label}: {url}") from exc
  
  def expect_status(url: str, label: str, status_code: int) -> None:
      print(f"EXPECT {status_code} {label}: {url}")
      try:
          urllib.request.urlopen(url, timeout=90)
      except urllib.error.HTTPError as exc:
          assert exc.code == status_code
          return
      raise AssertionError(f"expected HTTP {status_code} at {label}: {url}")
